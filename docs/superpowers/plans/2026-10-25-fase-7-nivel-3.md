```ts
import type { Metric, TimeseriesPoint } from '../../shared/api/types';
import { CUMULATIVE_LABELS, NEW_LABELS, formatDate, formatNumber } from '../../shared/format/format';
import { addDays, periodRange, type DateRange, type Period } from '../../shared/period/period';

export const AVERAGE_WINDOW = 7;

export type Scale = 'linear' | 'log';

export interface ChartRanges {
  visible: DateRange;
  fetch: { from: string; to: string };
}

// The fetch starts 6 days earlier so the 7-day average is complete on the first visible day.
export function chartRanges(period: Period, coverage: { from: string; to: string }): ChartRanges {
  const visible = periodRange(period, coverage);
  const start = addDays(visible.from, -(AVERAGE_WINDOW - 1));
  return {
    visible,
    fetch: { from: start < coverage.from ? coverage.from : start, to: visible.to },
  };
}

export interface DailyNewRow {
  date: string;
  value: number;
  bar: number;
  average: number | null;
}

// RN-07: `value` keeps what the source sent; `bar` and `average` are never negative.
export function dailyNewRows(
  points: readonly TimeseriesPoint[],
  visibleFrom: string,
): DailyNewRow[] {
  const byDate = new Map(points.map((point) => [point.date, point.new]));
  const averageAt = (date: string): number | null => {
    let sum = 0;
    for (let back = 0; back < AVERAGE_WINDOW; back += 1) {
      const value = byDate.get(addDays(date, -back));
      if (value === undefined) return null;
      sum += value;
    }
    return Math.max(0, Math.round(sum / AVERAGE_WINDOW));
  };
  return points
    .filter((point) => point.date >= visibleFrom)
    .map((point) => ({
      date: point.date,
      value: point.new,
      bar: Math.max(0, point.new),
      average: averageAt(point.date),
    }));
}

// A log axis has no place for zero or negatives: those days become gaps.
export function scaled(value: number | null | undefined, scale: Scale): number | null {
  if (value === null || value === undefined) return null;
  return scale === 'log' && value <= 0 ? null : value;
}

export interface CumulativeRow {
  date: string;
  world: number | null;
  country: number | null;
}

export function cumulativeRows(
  world: readonly TimeseriesPoint[],
  country: readonly TimeseriesPoint[] | null,
  scale: Scale,
): CumulativeRow[] {
  const countryByDate = new Map((country ?? []).map((point) => [point.date, point.total]));
  return world.map((point) => ({
    date: point.date,
    world: scaled(point.total, scale),
    country: country === null ? null : scaled(countryByDate.get(point.date), scale),
  }));
}

export interface SeriesEnds {
  name: string;
  first: number;
  last: number;
}

export function seriesEnds(name: string, points: readonly TimeseriesPoint[]): SeriesEnds | null {
  const first = points[0];
  const last = points.at(-1);
  return first && last ? { name, first: first.total, last: last.total } : null;
}

function rangeText(range: { from: string; to: string }): string {
  return `de ${formatDate(range.from)} a ${formatDate(range.to)}`;
}

export function cumulativeSummary(
  metric: Metric,
  range: { from: string; to: string },
  series: readonly SeriesEnds[],
): string {
  const parts = series.map(
    (item) => `${item.name}: de ${formatNumber(item.first)} para ${formatNumber(item.last)}`,
  );
  return `${CUMULATIVE_LABELS[metric]} ${rangeText(range)} — ${parts.join('; ')}.`;
}

export function dailyNewSummary(
  metric: Metric,
  place: string,
  range: { from: string; to: string },
  rows: readonly DailyNewRow[],
): string {
  const [first, ...rest] = rows;
  if (!first) return '';
  // The first day wins a tie.
  const peak = rest.reduce((best, row) => (row.value > best.value ? row : best), first);
  const lastAverage = rows.at(-1)?.average ?? null;
  return `${NEW_LABELS[metric]} (${place}) ${rangeText(range)}: pico de ${formatNumber(peak.value)} em ${formatDate(peak.date)}; média de 7 dias no último dia: ${formatNumber(lastAverage)}.`;
}
```

import type { ReactNode } from 'react';
import { isApiError } from '../../shared/api/client';
import { EmptyState } from '../../shared/ui/empty-state';
import { ErrorState } from '../../shared/ui/error-state';
import { Skeleton } from '../../shared/ui/skeleton';

export const NO_DATA = 'Sem dados no período';
const NOT_LOADED = 'Dados ainda não carregados';

interface QueryState {
  isPending: boolean;
  isError: boolean;
  error: unknown;
  refetch: () => Promise<unknown>;
}

interface SeriesFallbackOptions {
  meta: QueryState;
  ready: boolean;
  // Only enabled queries: a disabled one stays pending forever.
  series: readonly QueryState[];
  loadingLabel: string;
  errorMessage: string;
  height?: string;
}

// Loading, empty and error states of the charts, with the time card's rules; null once every
// series has loaded.
export function seriesFallback({
  meta,
  ready,
  series,
  loadingLabel,
  errorMessage,
  height = '15rem',
}: SeriesFallbackOptions): ReactNode {
  const loading = <Skeleton label={loadingLabel} height={height} />;
  if (meta.isPending) return loading;
  if (meta.isError) {
    return <ErrorState message={errorMessage} onRetry={() => void meta.refetch()} />;
  }
  if (!ready) return <EmptyState message={NOT_LOADED} />;
  const failed = series.find((query) => query.isError);
  if (failed) {
    if (isApiError(failed.error, 'DATA_NOT_READY')) return <EmptyState message={NOT_LOADED} />;
    // The page drops an unknown country and the chart falls back to the world.
    if (isApiError(failed.error, 'COUNTRY_NOT_FOUND')) return <EmptyState message={NO_DATA} />;
    return (
      <ErrorState
        message={errorMessage}
        onRetry={() => {
          // A stale coverage would ask for dates out of range: refresh it first.
          void meta.refetch().then(() => Promise.all(series.map((query) => query.refetch())));
        }}
      />
    );
  }
  if (series.some((query) => query.isPending)) return loading;
  return null;
}

  expect(weekly).toHaveLength(0);
