```sql
-- Raw SQL (not modeled by Prisma): CHECK constraints and the global materialized view.
ALTER TABLE "country"
  ADD CONSTRAINT "country_coordinates_required" CHECK ("kind" <> 'country' OR ("lat" IS NOT NULL AND "lng" IS NOT NULL)),
  ADD CONSTRAINT "country_lat_range" CHECK ("lat" BETWEEN -90 AND 90),
  ADD CONSTRAINT "country_lng_range" CHECK ("lng" BETWEEN -180 AND 180);

ALTER TABLE "country_alias"
  ADD CONSTRAINT "country_alias_mapped_has_country" CHECK (("status" = 'mapped') = ("country_code" IS NOT NULL));

ALTER TABLE "daily_stat"
  ADD CONSTRAINT "daily_stat_total_non_negative" CHECK ("total" >= 0);

CREATE MATERIALIZED VIEW "global_daily_stat" AS
SELECT "source_id", "metric", "stat_date",
       sum("total")::bigint     AS "total",
       sum("new_value")::bigint AS "new_value",
       count(*)::int            AS "country_count"
FROM "daily_stat"
GROUP BY "source_id", "metric", "stat_date";

CREATE UNIQUE INDEX "global_daily_stat_key" ON "global_daily_stat" ("source_id", "metric", "stat_date");
```

```ts
import { z } from 'zod';

const postgresUrl = z
  .string()
  .regex(/^postgres(ql)?:\/\//, 'must be a postgres:// or postgresql:// URL');

const dbEnvSchema = z.object({
  DATABASE_URL: postgresUrl,
  DATABASE_URL_SYNC: postgresUrl,
});

export type DbEnv = z.infer<typeof dbEnvSchema>;

export function loadDbEnv(source: NodeJS.ProcessEnv = process.env): DbEnv {
  const result = dbEnvSchema.safeParse(source);
  if (!result.success) {
    throw new Error(`Invalid database environment variables:\n${z.prettifyError(result.error)}`);
  }
  return result.data;
}
```

- [ ] **Step 4: Implementar `api/prisma/roles-sql.ts`**

```ts
import pg from 'pg';

const { escapeIdentifier, escapeLiteral } = pg;

export interface ReaderRoleOptions {
  owner: string;
  reader: string;
  password: string;
  database: string;
  readerExists: boolean;
}

export interface ConnectionParts {
  user: string;
  password: string;
  database: string;
}

export function buildReaderRoleSql(options: ReaderRoleOptions): string[] {
  const owner = escapeIdentifier(options.owner);
  const reader = escapeIdentifier(options.reader);
  const password = escapeLiteral(options.password);
  const database = escapeIdentifier(options.database);

  return [
    `${options.readerExists ? 'ALTER' : 'CREATE'} ROLE ${reader} WITH LOGIN PASSWORD ${password}`,
    `GRANT CONNECT ON DATABASE ${database} TO ${reader}`,
    `GRANT USAGE ON SCHEMA public TO ${reader}`,
    `GRANT SELECT ON ALL TABLES IN SCHEMA public TO ${reader}`,
    `ALTER DEFAULT PRIVILEGES FOR ROLE ${owner} IN SCHEMA public GRANT SELECT ON TABLES TO ${reader}`,
  ];
}

export function parseConnectionUrl(url: string): ConnectionParts {
  const parsed = new URL(url);
  const user = decodeURIComponent(parsed.username);
  const password = decodeURIComponent(parsed.password);
  const database = decodeURIComponent(parsed.pathname.replace(/^\//, ''));
  if (!user || !password || !database) {
    throw new Error('Connection URL must include user, password and database');
  }
  return { user, password, database };
}
```

- [ ] **Step 6: Implementar o CLI `api/prisma/roles.ts`**

```ts
import pg from 'pg';
import { loadDbEnv } from '../src/config/db-env.js';
import { createLogger } from '../src/lib/logger.js';
import { buildReaderRoleSql, parseConnectionUrl } from './roles-sql.js';

const env = loadDbEnv();
const logger = createLogger('info');
const reader = parseConnectionUrl(env.DATABASE_URL);
const owner = parseConnectionUrl(env.DATABASE_URL_SYNC);

const client = new pg.Client({ connectionString: env.DATABASE_URL_SYNC });
await client.connect();
try {
  const existing = await client.query('SELECT 1 FROM pg_roles WHERE rolname = $1', [reader.user]);
  const statements = buildReaderRoleSql({
    owner: owner.user,
    reader: reader.user,
    password: reader.password,
    database: reader.database,
    readerExists: existing.rowCount === 1,
  });
  await client.query('BEGIN');
  for (const statement of statements) {
    await client.query(statement);
  }
  await client.query('COMMIT');
  logger.info({ role: reader.user, database: reader.database }, 'reader role ready');
} catch (error) {
  await client.query('ROLLBACK').catch(() => undefined);
  throw error;
} finally {
```

```ts
export type CountryKind = 'country' | 'non_geographic';

export interface CountrySeed {
  code: string;
  name: string;
  kind: CountryKind;
  lat: number | null;
  lng: number | null;
  population: number | null;
  sourceName: string;
}

const NON_GEOGRAPHIC: Readonly<Record<string, { code: string; name: string }>> = {
  'Diamond Princess': { code: 'XDP', name: 'Navio Diamond Princess' },
  'MS Zaandam': { code: 'XMZ', name: 'Navio MS Zaandam' },
  'Summer Olympics 2020': { code: 'XSO', name: 'Jogos Olímpicos de Verão 2020' },
  'Winter Olympics 2022': { code: 'XWO', name: 'Jogos Olímpicos de Inverno 2022' },
};

const NAME_OVERRIDES: Readonly<Record<string, string>> = {
  CD: 'República Democrática do Congo',
  MM: 'Mianmar',
};

const regionNames = new Intl.DisplayNames(['pt-BR'], { type: 'region', fallback: 'none' });

export function parseCsv(text: string): string[][] {
  const input = text.replace(/^﻿/, '');
  const rows: string[][] = [];
  let row: string[] = [];
  let field = '';
  let inQuotes = false;

  for (let i = 0; i < input.length; i += 1) {
    const char = input.charAt(i);
    if (inQuotes) {
      if (char === '"' && input.charAt(i + 1) === '"') {
        field += '"';
        i += 1;
      } else if (char === '"') {
        inQuotes = false;
      } else {
        field += char;
      }
    } else if (char === '"') {
      inQuotes = true;
    } else if (char === ',') {
      row.push(field);
      field = '';
    } else if (char === '\n' || char === '\r') {
      if (char === '\r' && input.charAt(i + 1) === '\n') {
        i += 1;
      }
      row.push(field);
      rows.push(row);
      row = [];
      field = '';
    } else {
      field += char;
    }
  }

  if (field !== '' || row.length > 0) {
    row.push(field);
    rows.push(row);
  }
  return rows;
}

interface Columns {
  iso2: number;
  iso3: number;
  admin2: number;
  province: number;
  country: number;
  lat: number;
  lng: number;
  population: number;
}

function cell(row: string[], index: number): string {
  return row[index] ?? '';
}

function coordinate(value: string, limit: number, sourceName: string): number {
  const parsed = Number(value);
  if (value === '' || !Number.isFinite(parsed) || Math.abs(parsed) > limit) {
    throw new Error(`Invalid coordinate "${value}" for ${sourceName}`);
  }
  return parsed;
}

function population(value: string, sourceName: string): number | null {
  if (value === '') {
    return null;
  }
  const parsed = Number(value);
  if (!Number.isSafeInteger(parsed) || parsed < 0) {
    throw new Error(`Invalid population "${value}" for ${sourceName}`);
  }
  return parsed;
}

function toSeed(row: string[], columns: Columns): CountrySeed {
  const sourceName = cell(row, columns.country);
  const iso3 = cell(row, columns.iso3);

  if (iso3 === '') {
    const entity = NON_GEOGRAPHIC[sourceName];
    if (!entity) {
      throw new Error(`Row without iso3 and no non-geographic mapping: ${sourceName}`);
    }
    return { ...entity, kind: 'non_geographic', lat: null, lng: null, population: null, sourceName };
  }

  const iso2 = cell(row, columns.iso2);
  const name = NAME_OVERRIDES[iso2] ?? regionNames.of(iso2);
  if (!name) {
    throw new Error(`No pt-BR name for ${iso2} (${sourceName})`);
  }
  return {
    code: iso3,
    name,
    kind: 'country',
    lat: coordinate(cell(row, columns.lat), 90, sourceName),
    lng: coordinate(cell(row, columns.lng), 180, sourceName),
    population: population(cell(row, columns.population), sourceName),
    sourceName,
  };
}

export function buildCountrySeeds(csvText: string): CountrySeed[] {
  const [header, ...rows] = parseCsv(csvText).filter((row) => row.some((value) => value !== ''));
  if (!header) {
    throw new Error('Empty reference CSV');
  }
  const column = (name: string): number => {
    const index = header.indexOf(name);
    if (index === -1) {
      throw new Error(`Missing column ${name}`);
    }
    return index;
  };
  const columns: Columns = {
    iso2: column('iso2'),
    iso3: column('iso3'),
    admin2: column('Admin2'),
    province: column('Province_State'),
    country: column('Country_Region'),
    lat: column('Lat'),
    lng: column('Long_'),
    population: column('Population'),
  };

  const seeds = rows
    .filter((row) => cell(row, columns.province) === '' && cell(row, columns.admin2) === '')
    .map((row) => toSeed(row, columns));

  const seen = new Set<string>();
  for (const seed of seeds) {
    if (seen.has(seed.code)) {
      throw new Error(`Duplicate country code ${seed.code}`);
    }
    seen.add(seed.code);
  }
  return seeds.sort((a, b) => a.code.localeCompare(b.code));
}
```

```ts
import { readFileSync } from 'node:fs';
import { PrismaPg } from '@prisma/adapter-pg';
import { loadDbEnv } from '../../src/config/db-env.js';
import { PrismaClient } from '../../src/generated/prisma/client.js';
import { createLogger } from '../../src/lib/logger.js';
import type { CountrySeed } from './reference.js';

const API_NINJAS_SOURCE = {
  code: 'api_ninjas',
  name: 'API Ninjas — COVID-19',
  url: 'https://api-ninjas.com/api/covid19',
  license:
    'Termos de uso da API Ninjas (plano gratuito: exige atribuição; não permite cache nem armazenamento)',
  attribution: 'Dados: API Ninjas (fonte original: JHU CSSE)',
};

const env = loadDbEnv();
const logger = createLogger('info');
const countries = JSON.parse(
  readFileSync(new URL('./countries.json', import.meta.url), 'utf8'),
) as CountrySeed[];

const prisma = new PrismaClient({
  adapter: new PrismaPg({ connectionString: env.DATABASE_URL_SYNC }),
});

try {
  await prisma.$transaction(
    async (tx) => {
      const source = await tx.dataSource.upsert({
        where: { code: API_NINJAS_SOURCE.code },
        create: API_NINJAS_SOURCE,
        update: API_NINJAS_SOURCE,
      });

      for (const country of countries) {
        const data = {
          name: country.name,
          kind: country.kind,
          lat: country.lat,
          lng: country.lng,
          population: country.population === null ? null : BigInt(country.population),
        };
        await tx.country.upsert({
          where: { code: country.code },
          create: { code: country.code, ...data },
          update: data,
        });
        await tx.countryAlias.upsert({
          where: { sourceId_sourceName: { sourceId: source.id, sourceName: country.sourceName } },
          create: {
            sourceId: source.id,
            sourceName: country.sourceName,
            countryCode: country.code,
            status: 'mapped',
          },
          update: { countryCode: country.code, status: 'mapped' },
        });
      }
    },
    { timeout: 60_000 },
  );
  logger.info({ countries: countries.length, source: API_NINJAS_SOURCE.code }, 'seed complete');
} finally {
  await prisma.$disconnect();
}
```

```ts
export interface CountryFeatureProperties {
  iso3: string;
  name: string;
}

export interface SourceProperties {
  ISO_A3_EH?: unknown;
  ADM0_A3?: unknown;
  NAME?: unknown;
}

export interface Feature<P> {
  type: 'Feature';
  properties: P;
  geometry: unknown;
}

export interface FeatureCollection<P> {
  type: 'FeatureCollection';
  features: Feature<P>[];
}

const CODE_OVERRIDES: Readonly<Record<string, string>> = {
  KOS: 'XKS',
};

export function toCountryGeo(
  source: FeatureCollection<SourceProperties>,
): FeatureCollection<CountryFeatureProperties> {
  return {
    type: 'FeatureCollection',
    features: source.features.map((feature) => {
      const { ISO_A3_EH, ADM0_A3, NAME } = feature.properties;
      const code = ISO_A3_EH === '-99' ? ADM0_A3 : ISO_A3_EH;
      if (typeof code !== 'string' || typeof NAME !== 'string') {
        throw new Error(`Feature without a usable code or name: ${JSON.stringify(NAME)}`);
      }
      return {
        type: 'Feature',
        properties: { iso3: CODE_OVERRIDES[code] ?? code, name: NAME },
        geometry: feature.geometry,
      };
    }),
  };
}
```
